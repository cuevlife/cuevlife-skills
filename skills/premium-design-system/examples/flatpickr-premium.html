Full JavaScript (copy-paste) 
document.addEventListener('DOMContentLoaded', function () {
    if (!window.flatpickr) return;

    var isThai = document.documentElement.lang === 'th';

    function createPremiumSelect(container, options, initialValue, onSelect) {
        var select = document.createElement('div');
        select.className = 'sb-premium-select';
        var trigger = document.createElement('div');
        trigger.className = 'sb-premium-trigger';
        select.appendChild(trigger);
        var label = document.createElement('span');
        label.className = 'sb-premium-label';
        trigger.appendChild(label);
        var chevron = document.createElement('i');
        chevron.className = 'bi bi-chevron-down sb-premium-chevron';
        trigger.appendChild(chevron);
        var menu = document.createElement('div');
        menu.className = 'sb-premium-menu custom-scrollbar';
        select.appendChild(menu);
        var currentValue = initialValue;
        function update(newVal) {
            currentValue = newVal;
            var opt = options.find(function(o) { return o.v == currentValue; });
            label.textContent = opt ? opt.l : '';
            menu.innerHTML = '';
            options.forEach(function(opt) {
                var item = document.createElement('div');
                item.className = 'sb-premium-item' + (opt.v == currentValue ? ' selected' : '');
                item.innerHTML = '<span>' + opt.l + '</span>' + (opt.v == currentValue ? '<i class="bi bi-check2 text-xs"></i>' : '');
                item.onmousedown = function(e) { e.preventDefault(); e.stopPropagation(); onSelect(opt.v); update(opt.v); menu.classList.remove('active'); trigger.classList.remove('active'); };
                menu.appendChild(item);
            });
        }
        trigger.onmousedown = function(e) {
            e.preventDefault(); e.stopPropagation();
            document.querySelectorAll('.sb-premium-menu').forEach(function(m) { if (m !== menu) m.classList.remove('active'); });
            document.querySelectorAll('.sb-premium-trigger').forEach(function(t) { if (t !== trigger) t.classList.remove('active'); });
            var isActive = menu.classList.toggle('active');
            trigger.classList.toggle('active', isActive);
            if (isActive) {
                setTimeout(function() {
                    var selectedItem = menu.querySelector('.sb-premium-item.selected');
                    if (selectedItem) selectedItem.scrollIntoView({ block: 'nearest', behavior: 'auto' });
                }, 50);
            }
        };
        update(initialValue);
        container.appendChild(select);
        return { dom: select, update: update };
    }

    flatpickr('.js-date', {
        dateFormat: 'Y-m-d',
        locale: isThai && flatpickr.l10ns && flatpickr.l10ns.th ? flatpickr.l10ns.th : 'en',
        monthSelectorType: 'static',
        allowInput: false,
        disableMobile: true,
        onReady: function(selectedDates, dateStr, instance) {
            var monthContainer = instance.calendarContainer.querySelector('.flatpickr-month');
            if (!monthContainer) return;
            var header = document.createElement('div');
            header.className = 'sb-premium-header';
            var prevBtn = document.createElement('div');
            prevBtn.className = 'sb-premium-nav prev';
            prevBtn.innerHTML = '<i class="bi bi-chevron-left"></i>';
            prevBtn.onmousedown = function(e) { e.preventDefault(); e.stopPropagation(); instance.changeMonth(-1, true); };
            header.appendChild(prevBtn);
            var monthNames = instance.l10n.months.longhand;
            var monthOpts = monthNames.map(function(name, i) { return { v: i, l: name }; });
            var monthSelect = createPremiumSelect(header, monthOpts, instance.currentMonth, function(val) { instance.changeMonth(val, false); });
            var yearOpts = [];
            for (var y = 2000; y <= 2100; y++) { yearOpts.push({ v: y, l: isThai ? (y + 543) : y }); }
            var yearSelect = createPremiumSelect(header, yearOpts, instance.currentYear, function(val) { instance.changeYear(val); });
            var nextBtn = document.createElement('div');
            nextBtn.className = 'sb-premium-nav next';
            nextBtn.innerHTML = '<i class="bi bi-chevron-right"></i>';
            nextBtn.onmousedown = function(e) { e.preventDefault(); e.stopPropagation(); instance.changeMonth(1, true); };
            header.appendChild(nextBtn);
            monthContainer.appendChild(header);
            instance.config.onMonthChange = instance.config.onMonthChange || [];
            instance.config.onYearChange = instance.config.onYearChange || [];
            instance.config.onMonthChange.push(function() { monthSelect.update(instance.currentMonth); });
            instance.config.onYearChange.push(function() { yearSelect.update(instance.currentYear); });
            document.addEventListener('mousedown', function(e) {
                if (!e.target.closest('.sb-premium-select')) {
                    document.querySelectorAll('.sb-premium-menu').forEach(function(m) { m.classList.remove('active'); });
                    document.querySelectorAll('.sb-premium-trigger').forEach(function(t) { t.classList.remove('active'); });
                }
            });
        }
    });

    // Date range example
    flatpickr('#date-range', {
        mode: 'range',
        dateFormat: 'Y-m-d',
        locale: isThai && flatpickr.l10ns && flatpickr.l10ns.th ? flatpickr.l10ns.th : 'en',
        monthSelectorType: 'static',
        maxDate: new Date(),
        onReady: /* same onReady as above */
    });
});
